# Task-5 Calculator Challenge
print("Calculator Program")
n1 = float(input("Enter the first operand:"))
n2 = float(input("Enter the second operand:"))
operator = input("Enter the Operation to be perfomed(+,-,*,/): ")

if operator== '+':
 output = n1+n2
 print(f"The addition is {n1}+{n2} is:{round(output,3)}")
elif operator=='-':
  output = n1-n2
  print(f"The difference is {n1}-{n2}is:{round(output,3)}")
elif operator=='*':
  output = n1*n2
  print(f"The multiplication is {n1}*{n2} is:{round(output,3)}") 
elif operator=='/':
  output = n1/n2
  print(f"The division is  {n1}/{n2} is:{round(output,3)}")
